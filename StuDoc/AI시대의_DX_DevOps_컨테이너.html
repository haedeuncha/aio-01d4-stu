{
  "name": "py-agent-dev",
  "build": { "dockerfile": "Dockerfile" },
  "workspaceFolder": "/workspace",
  "postCreateCommand": "uv sync"
}

# .devcontainer/Dockerfile
FROM python:3.12-slim
RUN apt-get update && apt-get install -y git tmux \
 && pip install uv
WORKDIR /workspace

# inside the container
$ uv run pytest
$ claude          # or codex
