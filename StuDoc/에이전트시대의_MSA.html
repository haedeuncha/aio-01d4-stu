# services/order/app/main.py
import json, uuid, redis
from fastapi import FastAPI, Header

app = FastAPI()
r = redis.Redis(host="redis")

@app.post("/orders")
def create(body: dict,
           key: str = Header(alias="Idempotency-Key")):
    if not r.set(f"idem:{key}", 1, nx=True, ex=86400):
        return {"status": "duplicate"}
    oid = str(uuid.uuid4())
    # save PENDING order + outbox row
    evt = json.dumps({"id": oid, **body})
    r.xadd("order.created", {"data": evt})
    return {"id": oid, "status": "PENDING"}
